## Global Constraints

- Только стандартная библиотека Python в `vmlogs_cli/` и `tests/`; `requires-python = ">=3.9"`. Не использовать синтаксис новее 3.9: `match`, `X | Y` в аннотациях без `from __future__ import annotations`, `str.removeprefix` можно (3.9). В каждом модуле пакета — `from __future__ import annotations`; дженерики — `list[str]`, `dict[str, str]`, `tuple[...]`, `Optional[...]` из `typing`, коллекции — из `collections.abc`.
- В файлах репозитория, сообщениях коммитов, тестовых данных и примерах — никаких имён реальных хостов, кластеров, проектов и компаний. Примеры — `example.com`, `myserver`, `shop`, `api`.
- Сообщения коммитов — одна строка-тема без тела, без `Co-Authored-By` и упоминаний ИИ; формат `<тип>: <суть>` по-русски.
- Ошибки для пользователя — исключение `vmlogs_cli.errors.CliError`, печатаются как `vmlogs-cli: <текст>` в stderr, код выхода 1, без трейсбэка.
- Тексты для пользователя (ошибки, подсказки, справка) — по-русски.
- Формат маркера обрезки — публичный контракт: `… [truncated: N/M chars, get <stream_id>/<time>]`.
- Перед каждым коммитом: `uv run ruff format . && uv run ruff check --fix .`, затем тесты.
- Тесты запускаются из корня репозитория: `uv run python -m unittest discover -s tests -v`.

class FindEnvFileTest(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.root = Path(self.tmp.name).resolve()
        self.home = self.root / "home"
        self.project = self.root / "work" / "shop" / "src"

        self.env = {"HOME": str(self.home)}

        self.tmp.cleanup()

    def test_project_env_found_upwards(self):
        expected = write(self.root / "work" / ".vmlogs-cli" / "env", "")
        write(config_dir(self.env) / "env", "")
        self.assertEqual(find_env_file(self.env, self.project), expected)

    def test_falls_back_to_config_dir(self):
        expected = write(self.home / ".config" / "vmlogs-cli" / "env", "")

    def test_vmlogs_env_beats_project(self):

`vmlogs_cli/config.py`:

```python
"""Профиль: где лежит env-файл и что в нём."""

from __future__ import annotations

import re
from collections.abc import Mapping
from dataclasses import dataclass
from pathlib import Path
from typing import Optional

from .errors import CliError

TRANSPORTS = ("direct", "kubectl", "grafana")
KNOWN_KEYS = frozenset(
    {
        "VMLOGS_TRANSPORT",
        "VMLOGS_URL",
        "VMLOGS_KUBE_SERVICE",
        "VMLOGS_KUBECONFIG",
        "VMLOGS_KUBE_CONTEXT",
        "VMLOGS_SSH_HOST",
        "VMLOGS_GRAFANA_URL",
        "VMLOGS_DATASOURCE",
        "VMLOGS_HIDE_FIELDS",
        "VMLOGS_CONTEXT_FIELDS",
    }
)
REQUIRED = {
    "direct": "VMLOGS_URL",
    "kubectl": "VMLOGS_KUBE_SERVICE",
    "grafana": "VMLOGS_GRAFANA_URL",
}
_SERVICE = re.compile(r"^[^/\s]+/[^/:\s]+:[^/:\s]+$")


@dataclass
class Settings:
    path: Path
    transport: str
    url: str = ""
    kube_service: str = ""
    kubeconfig: str = ""
    kube_context: str = ""
    ssh_host: str = ""
    grafana_url: str = ""
    datasource: str = ""
    hide_fields: Optional[list[str]] = None
    context_fields: Optional[list[str]] = None


def config_dir(env: Mapping[str, str]) -> Path:
    """Каталог настроек, cookie и профиля браузера: ~/.config/vmlogs-cli."""
    return Path(env.get("HOME") or Path.home()) / ".config" / "vmlogs-cli"


def _must_exist(path: Path, source: str) -> Path:
    if not path.is_file():
        raise CliError(f"профиль {path} не найден ({source})")
    return path


def find_env_file(env: Mapping[str, str], cwd, explicit: Optional[str] = None) -> Path:
    """Первый найденный: --env, $VMLOGS_ENV, .vmlogs-cli/env вверх от cwd, ~/.config/vmlogs-cli/env."""
    if explicit:
        return _must_exist(Path(explicit).expanduser(), "--env")
    if env.get("VMLOGS_ENV"):
        return _must_exist(Path(env["VMLOGS_ENV"]).expanduser(), "VMLOGS_ENV")
    start = Path(cwd).resolve()
    for directory in (start, *start.parents):
        candidate = directory / ".vmlogs-cli" / "env"
        if candidate.is_file():
            return candidate
    fallback = config_dir(env) / "env"
    if fallback.is_file():
        return fallback
    raise CliError(
        f"нет профиля: не нашёл ни $VMLOGS_ENV, ни .vmlogs-cli/env вверх от {start}, "
        f"ни {fallback}. Пример профиля — README.md, раздел «Профили»"
    )


def parse_env_file(path: Path) -> dict[str, str]:
    """Строки KEY=VALUE; пустые и # пропускаются, `export ` и парные кавычки снимаются."""
    values: dict[str, str] = {}
    for number, raw in enumerate(path.read_text(encoding="utf-8").splitlines(), 1):
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        if line.startswith("export "):
            line = line[len("export ") :].lstrip()
        key, sep, value = line.partition("=")
        key, value = key.strip(), value.strip()
        if not sep or not key:
            raise CliError(f"{path}:{number}: нужна строка KEY=VALUE, а не «{raw}»")
        if len(value) >= 2 and value[0] == value[-1] and value[0] in "\"'":
            value = value[1:-1]
        values[key] = value
    return values


def _csv(value: Optional[str]) -> Optional[list[str]]:
    if value is None:
        return None
    return [part.strip() for part in value.split(",") if part.strip()]


def load_settings(env: Mapping[str, str], cwd, explicit: Optional[str] = None) -> Settings:
    path = find_env_file(env, cwd, explicit)
    values = parse_env_file(path)
    unknown = sorted(key for key in values if key not in KNOWN_KEYS)
    if unknown:
        raise CliError(
            f"{path}: неизвестные ключи {', '.join(unknown)}. Известные: {', '.join(sorted(KNOWN_KEYS))}"
        )
    transport = values.get("VMLOGS_TRANSPORT", "")
    if transport not in TRANSPORTS:
        raise CliError(
            f"{path}: VMLOGS_TRANSPORT должен быть одним из {', '.join(TRANSPORTS)}, а не «{transport}»"
        )
    required = REQUIRED[transport]
    if not values.get(required):
        raise CliError(f"{path}: для VMLOGS_TRANSPORT={transport} нужен {required}")
    service = values.get("VMLOGS_KUBE_SERVICE", "")
    if transport == "kubectl" and not _SERVICE.match(service):
        raise CliError(
            f"{path}: VMLOGS_KUBE_SERVICE должен быть вида <namespace>/<service>:<port>, а не «{service}»"
        )
    return Settings(
        path=path,
        transport=transport,
        url=values.get("VMLOGS_URL", "").rstrip("/"),
        kube_service=service,
        kubeconfig=values.get("VMLOGS_KUBECONFIG", ""),
        kube_context=values.get("VMLOGS_KUBE_CONTEXT", ""),
        ssh_host=values.get("VMLOGS_SSH_HOST", ""),
        grafana_url=values.get("VMLOGS_GRAFANA_URL", "").rstrip("/"),
        datasource=values.get("VMLOGS_DATASOURCE", ""),
        hide_fields=_csv(values.get("VMLOGS_HIDE_FIELDS")),
        context_fields=_csv(values.get("VMLOGS_CONTEXT_FIELDS")),
    )
```

`vmlogs_cli/timewin.py`:

```python
"""Окно времени запроса: длительность назад от «сейчас» или RFC3339 → RFC3339 UTC.

VictoriaLogs не понимает start=5m как «5 минут назад», поэтому окно считаем сами.
"""

from __future__ import annotations

import re
from datetime import datetime, timedelta, timezone
from typing import Optional

from .errors import CliError

_DURATION = re.compile(r"^(\d+)([smhdw])$")
_UNITS = {"s": "seconds", "m": "minutes", "h": "hours", "d": "days", "w": "weeks"}
_FRACTION = re.compile(r"\.(\d+)")


def _fmt(moment: datetime) -> str:
    return moment.astimezone(timezone.utc).isoformat().replace("+00:00", "Z")


def parse_point(value: str, now: datetime) -> str:
    """`15m`, `2h`, `1d`, `1w` — столько назад от now; иначе RFC3339 с часовым поясом."""
    match = _DURATION.match(value)
    if match:
        return _fmt(now - timedelta(**{_UNITS[match.group(2)]: int(match.group(1))}))
    # Python 3.9 понимает в дробной части секунд только 3 или 6 цифр, VictoriaLogs пишет 9.
    text = _FRACTION.sub(
        lambda f: "." + f.group(1)[:6].ljust(6, "0"), value.replace("Z", "+00:00"), count=1
    )
    try:
        moment = datetime.fromisoformat(text)
    except ValueError:
        raise CliError(
            f"не понял время «{value}»: нужна длительность (30m, 2h, 1d) или RFC3339"
        ) from None
    if moment.tzinfo is None:
        raise CliError(f"у времени «{value}» нет часового пояса: добавь Z или смещение (+03:00)")
    return _fmt(moment)


def window(
    since: str, start: Optional[str], end: Optional[str], now: datetime

"""Вывод: короткие строки для чтения агентом, JSONL, скрытие служебных полей."""

from __future__ import annotations

import json
from collections.abc import Iterable, Sequence
from typing import Optional

# Служебные поля сборщиков: у каждой записи их десятки, и они в разы больше самого
# сообщения. Окончание «.» — префикс. Меняется VMLOGS_HIDE_FIELDS, показать всё — --all-fields.
DEFAULT_HIDE = (
    "_stream",
    "_stream_id",
    "kubernetes.pod_labels.",
    "kubernetes.pod_annotations.",
    "kubernetes.container_id",
    "kubernetes.pod_ip",
    "kubernetes.pod_node_name",
    "k8s.pod_labels.",
    "k8s.pod_annotations.",
    "k8s.pod_ip",
    "k8s.pod_node_name",
    "k8s.pod_owner",
    "vector.",
    "source_type",
    "file",
    "offset",
    "partition",
    "topic",
    "@timestamp_end",
)
# Разные сборщики называют одно и то же по-разному: берём первое найденное.
CONTAINER_FIELDS = ("kubernetes.container_name", "k8s.container_name")
LEVEL_FIELDS = ("level", "json.level", "severity")


def is_hidden(name: str, hide: Sequence[str]) -> bool:
    return any(name.startswith(h) if h.endswith(".") else name == h for h in hide)


def is_record(row: dict) -> bool:
    """Запись лога, а не строка результата `| stats`."""
    return "_msg" in row


def _flat(value) -> str:
    text = value if isinstance(value, str) else json.dumps(value, ensure_ascii=False)
    return text.replace("\x00", "").replace("\r\n", "⏎").replace("\n", "⏎")


def _first(row: dict, names: Sequence[str]) -> tuple[Optional[str], str]:
    for name in names:
        value = row.get(name)
        if value not in (None, ""):
            return name, _flat(value)
    return None, ""


def _time_key(row: dict) -> tuple[str, str]:
    # Дробная часть разной длины: «…:54Z» раньше «…:54.1Z», а сравнение строк говорит обратное.
    moment = row.get("_time", "")
    return moment[:19], moment[19:].lstrip(".").rstrip("Z").ljust(9, "0")


def sort_records(rows: list[dict]) -> list[dict]:
    """Записи — по возрастанию времени; результат `| stats` — в порядке VictoriaLogs."""
    if rows and all(is_record(row) for row in rows):
        return sorted(rows, key=_time_key)
    return rows


def short_line(
    row: dict, width: int, hide: Sequence[str], context: Optional[Sequence[str]] = None
) -> str:
    """`<время> <контейнер> <УРОВЕНЬ> <сообщение>  k=v …`, длинное — с маркером обрезки."""
    if not is_record(row):
        return " ".join(f"{key}={_flat(value)}" for key, value in row.items())
    container_field, container = _first(row, CONTAINER_FIELDS)
    level_field, level = _first(row, LEVEL_FIELDS)
    used = {"_time", "_msg", container_field, level_field}
    if context is not None:
        tail = [(key, row[key]) for key in context if key in row and key not in used]
    else:
        tail = sorted(
            ((k, v) for k, v in row.items() if k not in used and not is_hidden(k, hide)),
            key=lambda kv: kv[0],
        )
    head = [part for part in (_flat(row.get("_time", "")), container, level.upper()) if part]
    line = " ".join([*head, _flat(row["_msg"])])
    if tail:
        line += "  " + " ".join(f"{key}={_flat(value)}" for key, value in tail)
    if width and len(line) > width:
        marker = f"truncated: {width}/{len(line)} chars"
        if row.get("_stream_id") and row.get("_time"):
            marker += f", get {row['_stream_id']}/{row['_time']}"
        line = f"{line[:width]}… [{marker}]"
    return line


def json_line(row: dict, width: int, hide: Sequence[str]) -> str:
    """Запись JSON без скрытых полей; строки длиннее width — с хвостом `…(+N)`."""
    out = {}
    for key, value in row.items():
        if is_hidden(key, hide):
            continue
        if isinstance(value, str):
            value = value.replace("\x00", "")
            if width and len(value) > width:
                value = f"{value[:width]}…(+{len(value) - width})"
        out[key] = value
    return json.dumps(out, ensure_ascii=False)


def field_lines(values: Iterable[dict], hide: Sequence[str]) -> tuple[list[str], int]:
    """Строки `hits<TAB>имя` по убыванию hits и число скрытых полей."""
    lines, hidden = [], 0
    for item in sorted(values, key=lambda v: (-int(v["hits"]), v["value"])):
        if is_hidden(item["value"], hide):
            hidden += 1
            continue
        lines.append(f"{item['hits']}\t{item['value']}")
    return lines, hidden
```

class FakeServer:
    """HTTP-сервер на свободном порту в фоновом потоке.

    handle(method, path, params, headers) -> (status, [(заголовок, значение)], тело).
    params — из query string и из тела формы вместе.
    """

    def __init__(self, handle):
        self.requests = []
        requests = self.requests

        class Handler(BaseHTTPRequestHandler):
            def _serve(self, method):
                parsed = urllib.parse.urlsplit(self.path)
                length = int(self.headers.get("Content-Length") or 0)
                body = self.rfile.read(length).decode() if length else ""
                params = dict(urllib.parse.parse_qsl(parsed.query))
                params.update(urllib.parse.parse_qsl(body))
                headers = {k.lower(): v for k, v in self.headers.items()}
                requests.append((method, parsed.path, params, headers))
                status, extra, payload = handle(method, parsed.path, params, headers)
                data = payload.encode() if isinstance(payload, str) else payload
                self.send_response(status)
                for key, value in extra:
                    self.send_header(key, value)
                self.send_header("Content-Length", str(len(data)))
                self.end_headers()
                self.wfile.write(data)

            def do_GET(self):
                self._serve("GET")

            def do_POST(self):
                self._serve("POST")

            def log_message(self, *args):
                pass

        self.server = ThreadingHTTPServer(("127.0.0.1", 0), Handler)
        self.url = f"http://127.0.0.1:{self.server.server_port}"
        self.thread = threading.Thread(target=self.server.serve_forever, daemon=True)
        self.thread.start()

    def close(self):
        self.server.shutdown()
        self.server.server_close()


def victorialogs(prefix=""):
    """Поддельная VictoriaLogs; prefix — путь перед /select (у прокси Grafana он свой)."""
    base = prefix + "/select/logsql/"

    def handle(method, path, params, headers):
        if not path.startswith(base):
            return 404, [], "not found"
        endpoint = path[len(base) :]
        query = params.get("query", "")
        if "bad(" in query:
            return 400, [], "cannot parse `query` arg: unexpected pipe; context: [bad(]"
        if endpoint == "query":
            if "nothing" in query:
                rows = []
            elif "| stats" in query:
                rows = [{"c": str(len(RECORDS))}]
            elif "_stream_id:" in query:
                rows = [RECORDS[1]]
            else:
                rows = RECORDS[: int(params.get("limit", "1000"))]
            return 200, [], "".join(json.dumps(row) + "\n" for row in rows)
        if endpoint == "field_names":
            values = [
                {"value": "_msg", "hits": 2},
                {"value": "_stream", "hits": 2},
                {"value": "level", "hits": 2},
            ]
            return 200, [], json.dumps({"values": values})
        if endpoint == "field_values":
            values = [{"value": "error", "hits": 1}, {"value": "info", "hits": 1}]
            return 200, [], json.dumps({"values": values})
        return 404, [], "unknown endpoint"

    return handle

Контракт для cli: describe() -> str, fetch(endpoint, params) -> Iterator[str] (строки ответа).

import json
import os
import re
import shlex
import subprocess
import urllib.error
import urllib.parse
import urllib.request
from collections.abc import Iterator

from .errors import CliError

DEFAULT_TIMEOUT = 120
# Тело ответа 400 kubectl показывает только с -v=8, строкой `"message": "<JSON-строка>"`.
_K8S_MESSAGE = re.compile(r'"message":\s*("(?:[^"\\]|\\.)*")')


class _NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, *args, **kwargs):
        return None


def no_redirect_opener() -> urllib.request.OpenerDirector:
    """Редирект — это SSO или переезд, а не данные: пусть приходит как HTTPError."""
    return urllib.request.build_opener(_NoRedirect)


def vl_error(code: int, body: str) -> str:
    if code == 400:
        return f"VictoriaLogs не принял запрос: {body[:2000]}"
    return f"HTTP {code}: {body[:1000]}"


def http_error_body(e: urllib.error.HTTPError) -> str:
    return e.read().decode("utf-8", errors="replace").strip()


def iter_lines(resp) -> Iterator[str]:
    """Непустые строки ответа по мере чтения; ответ закрывается в конце."""
    with resp:
        try:
            for raw in resp:
                line = raw.decode("utf-8", errors="replace").rstrip("\r\n")
                if line:
                    yield line
        except OSError as e:
            raise CliError(f"ответ оборвался: {e}") from None


class DirectTransport:
    """VictoriaLogs по адресу: port-forward, сервис в кластере, открытый адрес."""

    def __init__(self, url: str, timeout: int = DEFAULT_TIMEOUT):
        self.url = url.rstrip("/")
        self.timeout = timeout

    def describe(self) -> str:
        return f"direct {self.url}"

    def fetch(self, endpoint: str, params: dict[str, str]) -> Iterator[str]:
        data = urllib.parse.urlencode(params).encode()
        req = urllib.request.Request(f"{self.url}/select/logsql/{endpoint}", data=data)
        try:
            resp = no_redirect_opener().open(req, timeout=self.timeout)
        except urllib.error.HTTPError as e:
            raise CliError(vl_error(e.code, http_error_body(e))) from None
        except (urllib.error.URLError, OSError) as e:
            reason = getattr(e, "reason", e)
            raise CliError(f"{self.url} не ответил ({reason}) — проверь адрес и VPN") from None
        return iter_lines(resp)


def _remote_quote(part: str) -> str:
    # `~/` в пути kubeconfig должна раскрыть удалённая оболочка, поэтому тильда — без кавычек.
    if part.startswith("~/"):
        return "~/" + shlex.quote(part[2:])
    return shlex.quote(part)


class KubectlTransport:
    """VictoriaLogs через прокси сервиса в API Kubernetes: `kubectl get --raw`, локально или по ssh."""

    def __init__(
        self,
        service: str,
        kubeconfig: str = "",
        context: str = "",
        ssh_host: str = "",
        timeout: int = DEFAULT_TIMEOUT,
        run=subprocess.run,
    ):
        self.namespace, self.service = service.split("/", 1)
        self.kubeconfig = kubeconfig
        self.context = context
        self.ssh_host = ssh_host
        self.timeout = timeout
        self.run = run

    def describe(self) -> str:
        where = f"ssh {self.ssh_host}: " if self.ssh_host else ""
        return f"kubectl {where}{self.namespace}/{self.service}"

    def command(self, endpoint: str, params: dict[str, str], verbose: bool = False) -> list[str]:
        query = urllib.parse.urlencode(params, quote_via=urllib.parse.quote)
        path = (
            f"/api/v1/namespaces/{self.namespace}/services/{self.service}"
            f"/proxy/select/logsql/{endpoint}?{query}"
        )
        cmd = ["kubectl"]
        if verbose:
            cmd.append("-v=8")
        if self.kubeconfig:
            # По ssh путь — на удалённой машине, раскрывать ~ здесь нельзя.
            local = os.path.expanduser(self.kubeconfig)
            cmd += ["--kubeconfig", self.kubeconfig if self.ssh_host else local]
        if self.context:
            cmd += ["--context", self.context]
        cmd += ["get", "--raw", path]
        if self.ssh_host:
            # ssh отдаёт команду удалённой оболочке строкой: без кавычек `&` в пути ушёл бы в фон.
            remote = " ".join(_remote_quote(part) for part in cmd)
            return ["ssh", "-o", "BatchMode=yes", self.ssh_host, remote]
        return cmd

    def _run(self, cmd: list[str]) -> subprocess.CompletedProcess:
        try:
            return self.run(cmd, capture_output=True, timeout=self.timeout)
        except FileNotFoundError:
            raise CliError(f"{cmd[0]} не найден в PATH") from None
        except subprocess.TimeoutExpired:
            raise CliError(f"{cmd[0]} не ответил за {self.timeout} с") from None

    def fetch(self, endpoint: str, params: dict[str, str]) -> Iterator[str]:
        proc = self._run(self.command(endpoint, params))
        if proc.returncode == 0:
            text = proc.stdout.decode("utf-8", errors="replace")
            return iter([line for line in text.splitlines() if line.strip()])
        err = proc.stderr.decode("utf-8", errors="replace").strip()
        if self.ssh_host and proc.returncode == 255:
            raise CliError(f"ssh {self.ssh_host}: {err}\nПроверь доступ: ssh {self.ssh_host} true")
        if "(BadRequest)" in err:
            message = self._bad_request_message(endpoint, params)
            if message:
                raise CliError(vl_error(400, message))
        where = f"kubectl на {self.ssh_host}" if self.ssh_host else "kubectl"
        raise CliError(
            f"{where}: {err}\nПроверь kubeconfig и контекст: kubectl config current-context"
        )

    def _bad_request_message(self, endpoint: str, params: dict[str, str]) -> str:
        """Текст ошибки VictoriaLogs из тела ответа: kubectl показывает его только с -v=8."""
        proc = self._run(self.command(endpoint, params, verbose=True))
        match = _K8S_MESSAGE.search(proc.stderr.decode("utf-8", errors="replace"))
        return json.loads(match.group(1)) if match else ""

"""Сессия Grafana: cookie-файл, проверка, продление без браузера и вход через playwright-cli."""

from __future__ import annotations

import contextlib
import os
import tempfile
import urllib.error
import urllib.parse
import urllib.request
from pathlib import Path
from typing import Optional

from .errors import CliError
from .transport import no_redirect_opener


def cookie_file(cfg_dir: Path, base_url: str) -> Path:
    """Cookie одна на хост Grafana: один вход обслуживает все профили с этим хостом."""
    return cfg_dir / "cookies" / urllib.parse.urlsplit(base_url).netloc


def merge_cookies(header: str, set_cookies: list[str]) -> str:
    """Заголовок Cookie с новыми значениями из Set-Cookie; пустое значение — cookie удалена."""
    values: dict[str, str] = {}
    parts = [part.strip() for part in header.split(";")]
    parts += [line.split(";", 1)[0].strip() for line in set_cookies]
    for part in parts:
        name, sep, value = part.partition("=")
        if sep and name:
            values[name] = value
    return "; ".join(f"{name}={value}" for name, value in values.items() if value)


def save_secret(path: Path, text: str) -> None:
    """Атомарная запись с правами 600: клиент не прочтёт половину, чужие не прочтут вовсе."""
    path.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
    fd, tmp = tempfile.mkstemp(dir=path.parent, prefix=f".{path.name}.")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as fh:
            fh.write(text + "\n")
        os.chmod(tmp, 0o600)
        os.replace(tmp, path)
    except BaseException:
        with contextlib.suppress(FileNotFoundError):
            os.unlink(tmp)
        raise


class GrafanaSession:
    """Cookie сессии Grafana в файле: проверка через /api/user, продление ротацией токена."""

    def __init__(self, base_url: str, path: Path, timeout: int = 30):
        self.base_url = base_url.rstrip("/")
        self.path = path
        self.timeout = timeout

    def saved(self) -> str:
        try:
            return self.path.read_text(encoding="utf-8").strip()
        except OSError:
            return ""

    def header(self) -> str:
        header = self.saved()
        if not header:
            raise CliError(f"нет входа в {self.base_url}: выполни vmlogs-cli login")
        return header

    def _call(self, method: str, path: str, header: str):
        data = b"" if method == "POST" else None
        req = urllib.request.Request(self.base_url + path, data=data, method=method)
        req.add_header("Cookie", header)
        req.add_header("x-grafana-org-id", "1")
        try:
            with no_redirect_opener().open(req, timeout=self.timeout) as resp:
                return resp.status, resp.headers
        except urllib.error.HTTPError as e:
            return e.code, e.headers
        except OSError:
            return 0, None

    def valid(self, header: str) -> bool:
        return bool(header) and self._call("GET", "/api/user", header)[0] == 200

    def rotate(self, header: str) -> Optional[str]:
        """Новый заголовок после ротации токена сессии или None, если Grafana не продлила."""
        _, headers = self._call("POST", "/api/user/auth-tokens/rotate", header)
        if headers is None:
            return None
        fresh = merge_cookies(header, headers.get_all("Set-Cookie") or [])
        return fresh if self.valid(fresh) else None

    def refresh(self) -> bool:
        """True — в файле рабочая cookie: сохранённая жива или продлена без браузера."""
        header = self.saved()
        if not header:
            return False
        if self.valid(header):
            return True
        fresh = self.rotate(header)
        if not fresh:
            return False
        save_secret(self.path, fresh)
        return True

LOGS_DATASOURCE_TYPE = "victoriametrics-logs-datasource"


class GrafanaTransport:
    """VictoriaLogs через прокси датасорса Grafana; вход — cookie сессии (login.GrafanaSession)."""

    def __init__(self, base_url: str, datasource: str, session, timeout: int = DEFAULT_TIMEOUT):
        self.base_url = base_url.rstrip("/")
        self.datasource = datasource
        self.session = session
        self.timeout = timeout
        self._uid = None

    def describe(self) -> str:
        return f"grafana {self.base_url}, датасорс {self.datasource or 'не задан'}"

    def _open(self, path: str, data=None, retry: bool = True):
        body = urllib.parse.urlencode(data).encode() if data is not None else None
        req = urllib.request.Request(self.base_url + path, data=body)
        req.add_header("Cookie", self.session.header())
        req.add_header("x-grafana-org-id", "1")
        try:
            return no_redirect_opener().open(req, timeout=self.timeout)
        except urllib.error.HTTPError as e:
            # Сессия Grafana живёт недолго: один раз продлеваем без браузера и повторяем.
            if e.code == 401 and retry and self.session.refresh():
                return self._open(path, data, retry=False)
            raise CliError(self._explain(e)) from None
        except (urllib.error.URLError, OSError) as e:
            reason = getattr(e, "reason", e)
            raise CliError(f"{self.base_url} не ответил ({reason}) — проверь VPN") from None

    def _explain(self, e: urllib.error.HTTPError) -> str:
        body = http_error_body(e)
        if 300 <= e.code < 400:
            return (
                f"{self.base_url} отвечает {e.code} на {e.headers.get('Location')} — "
                "Grafana переехала или просит вход: vmlogs-cli login"
            )
        if e.code == 401:
            return f"401 {body[:200]}\nСессия Grafana не продлевается: vmlogs-cli login"
        return vl_error(e.code, body)

    def datasources(self) -> list[tuple[str, str]]:
        """(uid, имя) датасорсов VictoriaLogs из /api/frontend/settings."""
        with self._open("/api/frontend/settings") as resp:
            settings = json.load(resp)
        items = settings.get("datasources", {}).values()
        pairs = ((d["uid"], d["name"]) for d in items if d.get("type") == LOGS_DATASOURCE_TYPE)
        return sorted(pairs, key=lambda pair: pair[1])

    def uid(self) -> str:
        if self._uid is None:
            if not self.datasource:
                raise CliError(
                    "не задан датасорс: VMLOGS_DATASOURCE в профиле или --ds; "
                    "список — vmlogs-cli datasources"
                )
            pairs = self.datasources()
            found = [uid for uid, name in pairs if self.datasource in (uid, name)]
            if not found:
                names = ", ".join(name for _, name in pairs) or "ни одного"
                raise CliError(f"датасорс логов «{self.datasource}» не найден. Есть: {names}")
            self._uid = found[0]
        return self._uid

    def fetch(self, endpoint: str, params: dict[str, str]) -> Iterator[str]:
        path = f"/api/datasources/proxy/uid/{self.uid()}/select/logsql/{endpoint}"
        return iter_lines(self._open(path, params))
